
:: Change the file name from run_server.txt to run_server.bat

@echo off
title Laravel Server - DMS

:: Automatically switch to the folder where this .bat file is located
cd /d "%~dp0"

echo ---------------------------------------------------
echo  Starting Document Management System
echo  Access locally at: http://localhost:8000
echo ---------------------------------------------------

:: Using 0.0.0.0 ensures it broadcasts to any local IP automatically
php artisan serve --host=0.0.0.0 --port=8000

pause
